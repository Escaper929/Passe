# PRD & Technical Spec: 拟真胶片画廊装裱工具 (Exhibition Matting Studio)

## 1. 项目定位与核心设计哲学
本项目旨在开发一款**高度拟真、还原美术馆与暗房展览装裱质感**的胶片数字化照片加白边与装裱工具。
- **杜绝塑料感与廉价感**：不使用生硬的纯白边框（`#FFFFFF`），杜绝在画面中堆砌高信息密度的快门/光圈参数。
- **物理光学拟真**：严谨还原 45° 倒角白芯剖面（Bevel Cut）、卡纸压在相纸上的下落微阴影（Cast/Inset Shadow）、纯棉无酸纸张微噪点（Paper Noise Texture）。
- **画廊级构图与无墨钢印**：严格遵循经典加权底边算法（Bottom-Weighted Rule），提供矢量旁轴相机极简线稿与无墨立体钢印（Blind Deboss）微浮雕效果。
- **纯前端无损渲染**：基于 Canvas 2D 离屏渲染，保护用户照片隐私，支持超清大图（4K/8K/中画幅扫描件）无损导出。

---

## 2. 布局几何学与美学参数规范

### 2.1 卡纸配色基准
- **博物馆暖白 (Warm Museum White)**: `#F8F7F3`（默认推荐，模拟纯棉无酸卡纸）
- **暗房象牙色 (Classic Ivory)**: `#F4F0E6`（模拟传统暗房手工纤维纸）
- **当代冷灰 (Cool Gallery Grey)**: `#ECEEF0`（模拟工业感冷调无酸卡纸）
- **炭黑展厅 (Matte Carbon Black)**: `#1C1C1E`（模拟高反差哑光碳黑卡纸）

### 2.2 几何开窗与加权算法
- **基准边距比例 (`marginRatio`)**：以照片短边为基准，默认取值 `0.14 ~ 0.15`。
- **底边加权系数 (`bottomWeight`)**：默认取值 `1.25`（即底部边距比上/左/右宽 25%），消除人眼视觉下坠错觉。
- **动态分辨率系数 (`scaleFactor`)**：所有线宽、倒角宽度、阴影模糊半径必须绑定公式：
  $$\text{scaleFactor} = \frac{\min(\text{canvasWidth}, \text{canvasHeight})}{1200}$$
  确保导出 8K 打印件与 1000px 屏幕预览比例完全一致。

## 3. 核心装裱渲染引擎实现 (src/engine/GalleryFramingEngine.ts)

```typescript
export interface FrameConfig {
  matColor?: string;             // 卡纸底色 (默认暖白 "#F8F7F3")
  marginRatio?: number;          // 边距基准比例 (默认 0.14)
  bottomWeight?: number;         // 底部加权倍数 (默认 1.25)
  targetAspect?: number | null;  // 固定画框比例 (如 4/3, 5/4, 留空为自适应)
  paperTextureIntensity?: number;// 纸张噪点强度 (0.01 ~ 0.08, 默认 0.04)
  bevelWidth?: number;           // 斜切切面基准宽度 (默认 2.5px)
  bevelColor?: string;           // 切芯颜色 (默认 "#FFFFFF")
  insetShadowBlur?: number;      // 卡纸落差内阴影半径 (默认 6px)
  enableStamp?: boolean;         // 是否启用无墨钢印 (默认 true)
  cameraModel?: string;          // 相机型号 (如 "LEICA M6")
  filmBrand?: string;            // 胶卷名称 (如 "KODAK PORTRA 400")
  stampDepth?: number;           // 钢印立体深度 (默认 1.2px)
}

export interface ExportOptions {
  format?: 'image/jpeg' | 'image/png';
  quality?: number;              // JPEG 质量 (0.1 ~ 1.0, 默认 0.95)
  maxDimension?: number;         // 限制最大导出边长
}

export class GalleryFramingEngine {
  private static noiseTileCache: HTMLCanvasElement | null = null;

  /** 生成无缝微噪点瓦片 (256x256 离屏单例) */
  private static getNoiseTile(intensity = 0.04): HTMLCanvasElement {
    if (this.noiseTileCache) return this.noiseTileCache;

    const size = 256;
    const tile = document.createElement('canvas');
    tile.width = size;
    tile.height = size;
    const ctx = tile.getContext('2d')!;
    const imgData = ctx.createImageData(size, size);
    const d = imgData.data;

    for (let i = 0; i < d.length; i += 4) {
      const n = (Math.random() - 0.5) * 255 * intensity;
      const v = Math.min(255, Math.max(0, 128 + n));
      d[i] = v;
      d[i + 1] = v;
      d[i + 2] = v;
      d[i + 3] = 255;
    }
    ctx.putImageData(imgData, 0, 0);
    this.noiseTileCache = tile;
    return tile;
  }

  /** 几何计算引擎：计算外框与开窗坐标 */
  public static calculateLayout(
    imgW: number,
    imgH: number,
    config: Required<Pick<FrameConfig, 'bottomWeight' 'marginRatio' 'targetAspect' |>>
  ) {
    const { marginRatio, bottomWeight, targetAspect } = config;
    const baseMargin = Math.min(imgW, imgH) * marginRatio;

    if (!targetAspect) {
      const mLeft = baseMargin;
      const mRight = baseMargin;
      const mTop = baseMargin;
      const mBottom = baseMargin * bottomWeight;
      return {
        canvasW: Math.round(imgW + mLeft + mRight),
        canvasH: Math.round(imgH + mTop + mBottom),
        x: Math.round(mLeft),
        y: Math.round(mTop),
        w: imgW,
        h: imgH,
        bottomOffset: mBottom,
      };
    }

    const minW = imgW + baseMargin * 2;
    const minH = imgH + baseMargin * (1 + bottomWeight);
    let canvasW = minW;
    let canvasH = minH;

    if (minW / minH > targetAspect) {
      canvasW = minW;
      canvasH = canvasW / targetAspect;
    } else {
      canvasH = minH;
      canvasW = canvasH * targetAspect;
    }

    const remX = canvasW - imgW;
    const remY = canvasH - imgH;
    const mLeft = remX / 2;
    const mTop = remY / (1 + bottomWeight);
    const mBottom = remY - mTop;

    return {
      canvasW: Math.round(canvasW),
      canvasH: Math.round(canvasH),
      x: Math.round(mLeft),
      y: Math.round(mTop),
      w: imgW,
      h: imgH,
      bottomOffset: mBottom,
    };
  }

  /** 经典旁轴相机简笔线稿矢量绘制 */
  private static drawCameraVector(ctx: CanvasRenderingContext2D, x: number, y: number, size: number) {
    ctx.save();
    ctx.translate(x, y);
    const s = size / 40;
    ctx.scale(s, s);

    ctx.beginPath();
    ctx.moveTo(-18, 10);
    ctx.lineTo(-18, -4);
    ctx.arcTo(-18, -7, -15, -7, 2);
    ctx.lineTo(10, -7);
    ctx.lineTo(11, -9);
    ctx.lineTo(17, -9);
    ctx.arcTo(19, -9, 19, -6, 2);
    ctx.lineTo(19, 10);
    ctx.arcTo(19, 12, 16, 12, 2);
    ctx.lineTo(-15, 12);
    ctx.arcTo(-18, 12, -18, 10, 2);
    ctx.closePath();
    ctx.stroke();

    ctx.strokeRect(-15, -10, 4, 3);
    ctx.strokeRect(12, -12, 4, 3);

    ctx.beginPath();
    ctx.arc(-1, 2.5, 7, 0, Math.PI * 2);
    ctx.stroke();
    ctx.beginPath();
    ctx.arc(-1, 2.5, 4.5, 0, Math.PI * 2);
    ctx.stroke();

    ctx.strokeRect(10, -4.5, 5, 3.5);
    ctx.fillRect(2, -4, 3, 2.5);

    ctx.restore();
  }

  /** 完整装裱渲染主流程 */
  public static render(
    image: HTMLImageElement | HTMLCanvasElement | ImageBitmap,
    config: FrameConfig = {},
    targetCanvas?: HTMLCanvasElement
  ): HTMLCanvasElement {
    const opts: Required<FrameConfig> = {
      matColor: '#F8F7F3',
      marginRatio: 0.14,
      bottomWeight: 1.25,
      targetAspect: null,
      paperTextureIntensity: 0.04,
      bevelWidth: 2.5,
      bevelColor: '#FFFFFF',
      insetShadowBlur: 6,
      enableStamp: true,
      cameraModel: 'LEICA M6',
      filmBrand: 'PORTRA 400',
      stampDepth: 1.2,
      ...config,
    };

    const layout = this.calculateLayout(image.width, image.height, opts);
    const canvas = targetCanvas || document.createElement('canvas');
    canvas.width = layout.canvasW;
    canvas.height = layout.canvasH;
    const ctx = canvas.getContext('2d')!;

    const scale = Math.min(layout.canvasW, layout.canvasH) / 1200;

    // 1. 卡纸纯色底色
    ctx.fillStyle = opts.matColor;
    ctx.fillRect(0, 0, layout.canvasW, layout.canvasH);

    // 2. 纸质微噪点 (EvenOdd 挖空相片区域)
    const noiseTile = this.getNoiseTile(opts.paperTextureIntensity);
    const pattern = ctx.createPattern(noiseTile, 'repeat');
    if (pattern) {
      ctx.save();
      ctx.beginPath();
      ctx.rect(0, 0, layout.canvasW, layout.canvasH);
      ctx.rect(layout.x, layout.y, layout.w, layout.h);
      ctx.clip('evenodd');

      ctx.globalCompositeOperation = 'soft-light';
      ctx.fillStyle = pattern;
      ctx.fillRect(0, 0, layout.canvasW, layout.canvasH);
      ctx.restore();
    }

    // 3. 45° 斜切白芯
    const bw = Math.max(1.5, Math.round(opts.bevelWidth * scale));
    ctx.fillStyle = opts.bevelColor;
    ctx.fillRect(layout.x - bw, layout.y - bw, layout.w + bw * 2, layout.h + bw * 2);

    ctx.fillStyle = 'rgba(255, 255, 255, 0.75)';
    ctx.fillRect(layout.x - bw, layout.y - bw, layout.w + bw * 2, bw);
    ctx.fillRect(layout.x - bw, layout.y - bw, bw, layout.h + bw * 2);

    ctx.fillStyle = 'rgba(0, 0, 0, 0.08)';
    ctx.fillRect(layout.x - bw, layout.y + layout.h, layout.w + bw * 2, bw);
    ctx.fillRect(layout.x + layout.w, layout.y - bw, bw, layout.h + bw * 2);

    // 4. 绘制原图主体
    ctx.drawImage(image, layout.x, layout.y, layout.w, layout.h);

    // 5. 卡纸下落内阴影 (Inset Shadow)
    ctx.save();
    ctx.beginPath();
    ctx.rect(layout.x, layout.y, layout.w, layout.h);
    ctx.clip();

    const shadowBlur = Math.max(3, Math.round(opts.insetShadowBlur * scale));
    const topGrad = ctx.createLinearGradient(0, layout.y, 0, layout.y + shadowBlur);
    topGrad.addColorStop(0, 'rgba(0, 0, 0, 0.22)');
    topGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');
    ctx.fillStyle = topGrad;
    ctx.fillRect(layout.x, layout.y, layout.w, shadowBlur);

    const leftGrad = ctx.createLinearGradient(layout.x, 0, layout.x + shadowBlur, 0);
    leftGrad.addColorStop(0, 'rgba(0, 0, 0, 0.16)');
    leftGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');
    ctx.fillStyle = leftGrad;
    ctx.fillRect(layout.x, layout.y, shadowBlur, layout.h);
    ctx.restore();

    // 6. 底部无墨钢印 (Blind Deboss)
    if (opts.enableStamp && (opts.cameraModel || opts.filmBrand)) {
      const centerX = layout.x + layout.w / 2;
      const iconY = Math.round(layout.y + layout.h + layout.bottomOffset * 0.36);
      const iconSize = Math.max(20, Math.round(26 * scale));
      const fontSize = Math.max(8, Math.round(9.5 * scale));
      const textY = iconY + Math.round(26 * scale);

      const drawStampShape = () => {
        ctx.lineCap = 'round';
        ctx.lineJoin = 'round';
        ctx.lineWidth = Math.max(1, 1.1 * scale);

        this.drawCameraVector(ctx, centerX, iconY, iconSize);

        const label = [opts.cameraModel, opts.filmBrand].filter(Boolean).join('   /   ').toUpperCase();
        ctx.save();
        ctx.font = `600 ${fontSize}px Inter, -apple-system, sans-serif`;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        if ('letterSpacing' in ctx) {
          // @ts-ignore
          ctx.letterSpacing = `${2.4 * scale}px`;
        }
        ctx.fillText(label, centerX, textY);
        ctx.restore();
      };

      const angle = Math.PI * 0.75;
      const offset = Math.max(0.8, opts.stampDepth * scale);
      const dx = Math.cos(angle) * offset;
      const dy = -Math.sin(angle) * offset;

      // 槽底压实
      ctx.save();
      ctx.globalCompositeOperation = 'multiply';
      ctx.fillStyle = 'rgba(0, 0, 0, 0.04)';
      ctx.strokeStyle = 'rgba(0, 0, 0, 0.04)';
      drawStampShape();
      ctx.restore();

      // 背光阴影
      ctx.save();
      ctx.translate(dx, dy);
      ctx.globalCompositeOperation = 'source-over';
      ctx.fillStyle = 'rgba(40, 35, 30, 0.30)';
      ctx.strokeStyle = 'rgba(40, 35, 30, 0.30)';
      drawStampShape();
      ctx.restore();

      // 迎光反光高光
      ctx.save();
      ctx.translate(-dx, -dy);
      ctx.globalCompositeOperation = 'screen';
      ctx.fillStyle = 'rgba(255, 255, 255, 0.72)';
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.72)';
      drawStampShape();
      ctx.restore();
    }

    return canvas;
  }

  /** 导出二进制 Blob */
  public static async exportBlob(
    image: HTMLImageElement | HTMLCanvasElement | ImageBitmap,
    config: FrameConfig = {},
    exportOpts: ExportOptions = {}
  ): Promise<Blob> {
    const { format = 'image/jpeg', quality = 0.95 } = exportOpts;
    const outputCanvas = this.render(image, config);

    return new Promise((resolve, reject) => {
      outputCanvas.toBlob(
        (blob) => {
          if (blob) resolve(blob);
          else reject(new Error('Canvas 导出失败'));
        },
        format,
        quality
      );
    });
  }

  /** 一键下载高分辨率大图 */
  public static async download(
    image: HTMLImageElement | HTMLCanvasElement | ImageBitmap,
    filename = 'framed-film-art.jpg',
    config: FrameConfig = {},
    exportOpts: ExportOptions = {}
  ): Promise<void> {
    const blob = await this.exportBlob(image, config, exportOpts);
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  }
}
