# PRD & 开发实现指南：Passe · 衬境 (Exhibition Matting Studio)

## 1. 项目定位与物理拟真规范

本项目开发一款面向胶片摄影爱好者的画廊级无损加白边与装裱工具，核心是还原实体展览装裱的光学层级与物理细节。

**装裱卡纸物理特性**：

- **配色标准**：博物馆暖白（#F8F7F3）、暗房象牙色（#F4F0E6）、当代冷灰（#ECEEF0）、炭黑展厅（#1C1C1E），杜绝纯白（#FFFFFF）塑料感。
- **纸质颗粒（Paper Texture）**：$256 \times 256$ 纯棉纸纤维微噪点平铺，采用 soft-light 混合模式与 EvenOdd 环形剪裁，只作用于卡纸，不污染相片画面。
- **45° 斜切白芯（Bevel Cut）**：卡纸开窗露出的倾斜剖面，左上受光面呈现细白高光，右下背光面呈现微弱暗收边。
- **相纸下落阴影（Inset Shadow）**：卡纸厚度压在相纸上产生的遮蔽暗影（Ambient Occlusion），限定在相纸范围内羽化。

**构图几何学**：

- **底边视觉加权（Weighted Bottom）**：底部边距默认比上、左、右宽 25%（系数 1.25），抵消人眼视平线下坠错觉。
- **动态尺寸无关性（Scale-Invariant）**：所有线宽、倒角、阴影半径绑定公式 $\text{scaleFactor} = \min(W, H) / 1200$，确保 1000px 网页预览与 8K 打印图比例绝对一致。
- **底部钢印徽标（Blind Deboss）**：放弃高密度曝光三要素，采用无墨凹版压印效果呈现经典旁轴相机矢量线稿与机型、胶卷大写字符。
- **光学原理**：左上槽底阴影 + 右下截光边缘反光 + 凹槽内部纸浆轻微压暗。

```typescript
export interface FrameConfig {
  matColor?: string;             // 卡纸底色 (默认 "#F8F7F3")
  marginRatio?: number;          // 边距基准比例 (默认 0.14)
  bottomWeight?: number;         // 底部加权倍数 (默认 1.25)
  targetAspect?: number | null;  // 固定外框比例 (如 4/3, 5/4, 留空为自适应)
  paperTextureIntensity?: number;// 纸张颗粒强度 (0.01 ~ 0.08, 默认 0.04)
  bevelWidth?: number;           // 45° 斜切切面基准宽度 (px, 默认 2.5)
  bevelColor?: string;           // 切芯颜色 (默认 "#FFFFFF")
  insetShadowBlur?: number;      // 卡纸下落内阴影半径 (默认 6)
  enableStamp?: boolean;         // 是否启用无墨钢印 (默认 true)
  cameraModel?: string;          // 相机型号 (如 "LEICA M6")
  filmBrand?: string;            // 胶卷名称 (如 "KODAK PORTRA 400")
  stampDepth?: number;           // 钢印下压深度 (默认 1.2)
}

export interface ExportOptions {
  format?: 'image/jpeg' | 'image/png';
  quality?: number;              // JPEG 质量 (0.1 ~ 1.0, 默认 0.98)
  maxDimension?: number;         // 导出长边最大像素限制
}

export class GalleryFramingEngine {
  private static noiseTileCache: HTMLCanvasElement | null = null;

  /**
   * 生成无缝微噪点瓦片（256x256 离屏单例）
   */
  private static getNoiseTile(intensity = 0.04): HTMLCanvasElement {
    if (this.noiseTileCache) return this.noiseTileCache;

    const size = 256;
    const tile = document.createElement('canvas');
    tile.width = size;
    tile.height = size;
    const ctx = tile.getContext('2d')!;
    const imgData = ctx.createImageData(size, size);
    const d = imgData.data;

    for (let i = 0; i < d.length; i += 4) {
      const n = (Math.random() - 0.5) * 255 * intensity;
      const v = Math.min(255, Math.max(0, 128 + n));
      d[i] = v;
      d[i + 1] = v;
      d[i + 2] = v;
      d[i + 3] = 255;
    }
    ctx.putImageData(imgData, 0, 0);
    this.noiseTileCache = tile;
    return tile;
  }

  /**
   * 几何计算引擎：计算外框与开窗坐标
   */
  public static calculateLayout(
    imgW: number,
    imgH: number,
    config: Required<Pick<FrameConfig, 'marginRatio' | 'bottomWeight' | 'targetAspect'>>
  ) {
    const { marginRatio, bottomWeight, targetAspect } = config;
    const baseMargin = Math.min(imgW, imgH) * marginRatio;

    if (!targetAspect) {
      const mLeft = baseMargin;
      const mRight = baseMargin;
      const mTop = baseMargin;
      const mBottom = baseMargin * bottomWeight;
      return {
        canvasW: Math.round(imgW + mLeft + mRight),
        canvasH: Math.round(imgH + mTop + mBottom),
        x: Math.round(mLeft),
        y: Math.round(mTop),
        w: imgW,
        h: imgH,
        bottomOffset: mBottom,
      };
    }

    const minW = imgW + baseMargin * 2;
    const minH = imgH + baseMargin * (1 + bottomWeight);
    let canvasW = minW;
    let canvasH = minH;

    if (minW / minH > targetAspect) {
      canvasW = minW;
      canvasH = canvasW / targetAspect;
    } else {
      canvasH = minH;
      canvasW = canvasH * targetAspect;
    }

    const remX = canvasW - imgW;
    const remY = canvasH - imgH;
    const mLeft = remX / 2;
    const mTop = remY / (1 + bottomWeight);
    const mBottom = remY - mTop;

    return {
      canvasW: Math.round(canvasW),
      canvasH: Math.round(canvasH),
      x: Math.round(mLeft),
      y: Math.round(mTop),
      w: imgW,
      h: imgH,
      bottomOffset: mBottom,
    };
  }

  /**
   * 矢量绘制：经典旁轴相机简笔线稿
   */
  private static drawCameraVector(ctx: CanvasRenderingContext2D, x: number, y: number, size: number) {
    ctx.save();
    ctx.translate(x, y);
    const s = size / 40;
    ctx.scale(s, s);

    ctx.beginPath();
    ctx.moveTo(-18, 10);
    ctx.lineTo(-18, -4);
    ctx.arcTo(-18, -7, -15, -7, 2);
    ctx.lineTo(10, -7);
    ctx.lineTo(11, -9);
    ctx.lineTo(17, -9);
    ctx.arcTo(19, -9, 19, -6, 2);
    ctx.lineTo(19, 10);
    ctx.arcTo(19, 12, 16, 12, 2);
    ctx.lineTo(-15, 12);
    ctx.arcTo(-18, 12, -18, 10, 2);
    ctx.closePath();
    ctx.stroke();

    ctx.strokeRect(-15, -10, 4, 3);
    ctx.strokeRect(12, -12, 4, 3);

    ctx.beginPath();
    ctx.arc(-1, 2.5, 7, 0, Math.PI * 2);
    ctx.stroke();
    ctx.beginPath();
    ctx.arc(-1, 2.5, 4.5, 0, Math.PI * 2);
    ctx.stroke();

    ctx.strokeRect(10, -4.5, 5, 3.5);
    ctx.fillRect(2, -4, 3, 2.5);

    ctx.restore();
  }

  /**
   * 完整装裱物理渲染管线
   */
  public static render(
    image: HTMLImageElement | HTMLCanvasElement | ImageBitmap,
    config: FrameConfig = {},
    targetCanvas?: HTMLCanvasElement
  ): HTMLCanvasElement {
    const opts: Required<FrameConfig> = {
      matColor: '#F8F7F3',
      marginRatio: 0.14,
      bottomWeight: 1.25,
      targetAspect: null,
      paperTextureIntensity: 0.04,
      bevelWidth: 2.5,
      bevelColor: '#FFFFFF',
      insetShadowBlur: 6,
      enableStamp: true,
      cameraModel: 'LEICA M6',
      filmBrand: 'PORTRA 400',
      stampDepth: 1.2,
      ...config,
    };

    const layout = this.calculateLayout(image.width, image.height, opts);
    const canvas = targetCanvas || document.createElement('canvas');
    canvas.width = layout.canvasW;
    canvas.height = layout.canvasH;
    const ctx = canvas.getContext('2d')!;

    const scale = Math.min(layout.canvasW, layout.canvasH) / 1200;

    // 1. 卡纸底色
    ctx.fillStyle = opts.matColor;
    ctx.fillRect(0, 0, layout.canvasW, layout.canvasH);

    // 2. 纸质微噪点 (EvenOdd 挖空照片区域，保持主体纯净)
    const noiseTile = this.getNoiseTile(opts.paperTextureIntensity);
    const pattern = ctx.createPattern(noiseTile, 'repeat');
    if (pattern) {
      ctx.save();
      ctx.beginPath();
      ctx.rect(0, 0, layout.canvasW, layout.canvasH);
      ctx.rect(layout.x, layout.y, layout.w, layout.h);
      ctx.clip('evenodd');

      ctx.globalCompositeOperation = 'soft-light';
      ctx.fillStyle = pattern;
      ctx.fillRect(0, 0, layout.canvasW, layout.canvasH);
      ctx.restore();
    }

    // 3. 45° 斜切白芯
    const bw = Math.max(1.5, Math.round(opts.bevelWidth * scale));
    ctx.fillStyle = opts.bevelColor;
    ctx.fillRect(layout.x - bw, layout.y - bw, layout.w + bw * 2, layout.h + bw * 2);

    // 斜切受光（顶、左：高光）
    ctx.fillStyle = 'rgba(255, 255, 255, 0.75)';
    ctx.fillRect(layout.x - bw, layout.y - bw, layout.w + bw * 2, bw);
    ctx.fillRect(layout.x - bw, layout.y - bw, bw, layout.h + bw * 2);

    // 斜切背光（底、右：微阴影）
    ctx.fillStyle = 'rgba(0, 0, 0, 0.08)';
    ctx.fillRect(layout.x - bw, layout.y + layout.h, layout.w + bw * 2, bw);
    ctx.fillRect(layout.x + layout.w, layout.y - bw, bw, layout.h + bw * 2);

    // 4. 照片主体
    ctx.drawImage(image, layout.x, layout.y, layout.w, layout.h);

    // 5. 卡纸下落内阴影 (Inset Shadow)
    ctx.save();
    ctx.beginPath();
    ctx.rect(layout.x, layout.y, layout.w, layout.h);
    ctx.clip();

    const shadowBlur = Math.max(3, Math.round(opts.insetShadowBlur * scale));
    const topGrad = ctx.createLinearGradient(0, layout.y, 0, layout.y + shadowBlur);
    topGrad.addColorStop(0, 'rgba(0, 0, 0, 0.22)');
    topGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');
    ctx.fillStyle = topGrad;
    ctx.fillRect(layout.x, layout.y, layout.w, shadowBlur);

    const leftGrad = ctx.createLinearGradient(layout.x, 0, layout.x + shadowBlur, 0);
    leftGrad.addColorStop(0, 'rgba(0, 0, 0, 0.16)');
    leftGrad.addColorStop(1, 'rgba(0, 0, 0, 0)');
    ctx.fillStyle = leftGrad;
    ctx.fillRect(layout.x, layout.y, shadowBlur, layout.h);
    ctx.restore();

    // 6. 底部无墨立体钢印 (Blind Deboss)
    if (opts.enableStamp && (opts.cameraModel || opts.filmBrand)) {
      const centerX = layout.x + layout.w / 2;
      const iconY = Math.round(layout.y + layout.h + layout.bottomOffset * 0.36);
      const iconSize = Math.max(20, Math.round(26 * scale));
      const fontSize = Math.max(8, Math.round(9.5 * scale));
      const textY = iconY + Math.round(26 * scale);

      const drawStampContent = () => {
        ctx.lineCap = 'round';
        ctx.lineJoin = 'round';
        ctx.lineWidth = Math.max(1, 1.1 * scale);

        this.drawCameraVector(ctx, centerX, iconY, iconSize);

        const label = [opts.cameraModel, opts.filmBrand].filter(Boolean).join('   /   ').toUpperCase();
        ctx.save();
        ctx.font = `600 ${fontSize}px Inter, -apple-system, sans-serif`;
        ctx.textAlign = 'left';
        ctx.textBaseline = 'middle';

        // 字距一律逐字排版，**不要**用 ctx.letterSpacing：
        // 原生实现把字距加在"每个字符之后"（末尾那个也加），而 textAlign:'center'
        // 是按"含尾随字距的推进宽度"居中的 —— 于是墨迹整体左偏 tracking/2
        // （实测字距 4px 时偏 2.00px），与上方那台按几何中心画的相机图标错开：
        // 1200px 预览偏 1.2px，8K 成品上到 8px。且 Safari 17 之前根本没有该属性。
        // 逐字累加、末尾不计字距，墨迹中心才落在请求中心上（±1px）。
        const tracking = 2.4 * scale;
        const glyphs = Array.from(label);
        if (glyphs.length > 0) {
          const advances = glyphs.map((glyph) => ctx.measureText(glyph).width);
          const inkWidth =
            advances.reduce((sum, width) => sum + width, 0) + tracking * (glyphs.length - 1);
          let cursor = centerX - inkWidth / 2;
          glyphs.forEach((glyph, index) => {
            ctx.fillText(glyph, cursor, textY);
            cursor += advances[index] + tracking;
          });
        }
        ctx.restore();
      };

      const angle = Math.PI * 0.75;
      const offset = Math.max(0.8, opts.stampDepth * scale);
      const dx = Math.cos(angle) * offset;
      const dy = -Math.sin(angle) * offset;

      // 槽底压暗
      ctx.save();
      ctx.globalCompositeOperation = 'multiply';
      ctx.fillStyle = 'rgba(0, 0, 0, 0.04)';
      ctx.strokeStyle = 'rgba(0, 0, 0, 0.04)';
      drawStampContent();
      ctx.restore();

      // 背光阴影棱
      ctx.save();
      ctx.translate(dx, dy);
      ctx.globalCompositeOperation = 'source-over';
      ctx.fillStyle = 'rgba(40, 35, 30, 0.30)';
      ctx.strokeStyle = 'rgba(40, 35, 30, 0.30)';
      drawStampContent();
      ctx.restore();

      // 迎光反光棱
      ctx.save();
      ctx.translate(-dx, -dy);
      ctx.globalCompositeOperation = 'screen';
      ctx.fillStyle = 'rgba(255, 255, 255, 0.72)';
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.72)';
      drawStampContent();
      ctx.restore();
    }

    return canvas;
  }

  /**
   * 导出为 Blob
   */
  public static async exportBlob(
    image: HTMLImageElement | HTMLCanvasElement | ImageBitmap,
    config: FrameConfig = {},
    exportOpts: ExportOptions = {}
  ): Promise<Blob> {
    const { format = 'image/jpeg', quality = 0.98, maxDimension } = exportOpts;

    let source = image;
    if (maxDimension && (image.width > maxDimension || image.height > maxDimension)) {
      const ratio = Math.min(maxDimension / image.width, maxDimension / image.height);
      const downCanvas = document.createElement('canvas');
      downCanvas.width = Math.round(image.width * ratio);
      downCanvas.height = Math.round(image.height * ratio);
      const dCtx = downCanvas.getContext('2d')!;
      dCtx.drawImage(image, 0, 0, downCanvas.width, downCanvas.height);
      source = downCanvas;
    }

    const outputCanvas = this.render(source, config);

    return new Promise((resolve, reject) => {
      outputCanvas.toBlob(
        (blob) => {
          if (blob) resolve(blob);
          else reject(new Error('Canvas 导出失败'));
        },
        format,
        quality
      );
    });
  }

  /**
   * 一键触发下载
   */
  public static async download(
    image: HTMLImageElement | HTMLCanvasElement | ImageBitmap,
    filename = 'framed-film-art.jpg',
    config: FrameConfig = {},
    exportOpts: ExportOptions = {}
  ): Promise<void> {
    const blob = await this.exportBlob(image, config, exportOpts);
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  }
}
```

```typescript
import React, { useState, useRef, useEffect, useCallback } from 'react';
import { GalleryFramingEngine, FrameConfig } from '../engine/GalleryFramingEngine';

const MAT_PRESETS = [
  { name: '博物馆暖白', color: '#F8F7F3' },
  { name: '暗房象牙色', color: '#F4F0E6' },
  { name: '当代冷灰', color: '#ECEEF0' },
  { name: '炭黑展厅', color: '#1C1C1E' },
];

const ASPECT_PRESETS = [
  { label: '自适应', value: null },
  { label: '4 : 3', value: 4 / 3 },
  { label: '5 : 4', value: 5 / 4 },
  { label: '1 : 1', value: 1 },
];

export const FramingStudio: React.FC = () => {
  const [sourceImage, setSourceImage] = useState<HTMLImageElement | null>(null);
  const [isExporting, setIsExporting] = useState(false);

  const [config, setConfig] = useState<FrameConfig>({
    matColor: '#F8F7F3',
    marginRatio: 0.15,
    bottomWeight: 1.25,
    targetAspect: null,
    bevelWidth: 2.5,
    enableStamp: true,
    cameraModel: 'LEICA M6',
    filmBrand: 'KODAK PORTRA 400',
    stampDepth: 1.2,
  });

  const previewCanvasRef = useRef<HTMLCanvasElement | null>(null);
  const rafIdRef = useRef<number | null>(null);

  // 防抖预览渲染
  const triggerRender = useCallback(() => {
    if (!sourceImage || !previewCanvasRef.current) return;

    if (rafIdRef.current) cancelAnimationFrame(rafIdRef.current);
    rafIdRef.current = requestAnimationFrame(() => {
      if (previewCanvasRef.current && sourceImage) {
        GalleryFramingEngine.render(sourceImage, config, previewCanvasRef.current);
      }
    });
  }, [sourceImage, config]);

  useEffect(() => {
    triggerRender();
    return () => {
      if (rafIdRef.current) cancelAnimationFrame(rafIdRef.current);
    };
  }, [triggerRender]);

  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = (event) => {
      const img = new Image();
      img.onload = () => setSourceImage(img);
      img.src = event.target?.result as string;
    };
    reader.readAsDataURL(file);
  };

  const handleExport = async () => {
    if (!sourceImage) return;
    try {
      setIsExporting(true);
      const filename = `Passe_${config.cameraModel || 'Gallery'}_${Date.now()}.jpg`;
      await GalleryFramingEngine.download(sourceImage, filename, config, {
        format: 'image/jpeg',
        quality: 0.98,
      });
    } catch (err) {
      console.error(err);
      alert('导出作品失败，请检查浏览器内存');
    } finally {
      setIsExporting(false);
    }
  };

  return (
    <div className="flex h-screen w-full bg-[#121212] text-[#E0E0E0] overflow-hidden font-sans">
      {/* 视口预览区 */}
      <main className="flex-1 flex items-center justify-center p-8 bg-[#18181A] relative select-none">
        {sourceImage ? (
          <div className="relative max-w-full max-h-full flex items-center justify-center shadow-2xl rounded-sm">
            <canvas
              ref={previewCanvasRef}
              className="max-h-[82vh] max-w-[82vw] object-contain transition-all duration-75"
              style={{ filter: 'drop-shadow(0 20px 30px rgba(0,0,0,0.5))' }}
            />
          </div>
        ) : (
          <label className="flex flex-col items-center justify-center w-96 h-64 border border-dashed border-[#333] hover:border-[#666] rounded-lg cursor-pointer bg-[#1e1e20] transition-colors">
            <span className="text-sm tracking-widest text-[#888] uppercase mb-2">选择或拖拽胶片扫描件</span>
            <span className="text-xs text-[#555]">支持无损全画幅/中画幅 JPEG, PNG</span>
            <input type="file" accept="image/*" className="hidden" onChange={handleFileUpload} />
          </label>
        )}
      </main>

      {/* 精密控制面板 */}
      <aside className="w-96 border-l border-[#242426] bg-[#161618] p-6 flex flex-col justify-between overflow-y-auto">
        <div className="space-y-6">
          <header>
            <h1 className="text-xs uppercase tracking-[0.25em] text-[#888]">Passe · 衬境</h1>
            <p className="text-base font-medium text-white mt-1">画廊装裱调校台</p>
          </header>

          {/* 卡纸材质 */}
          <div>
            <label className="text-xs text-[#777] uppercase tracking-wider block mb-2">卡纸材质 (Matboard)</label>
            <div className="grid grid-cols-2 gap-2">
              {MAT_PRESETS.map((p) => (
                <button
                  key={p.name}
                  onClick={() => setConfig({ ...config, matColor: p.color })}
                  className={`flex items-center gap-2 p-2 rounded text-xs border text-left transition-all ${
                    config.matColor === p.color
                      ? 'border-white bg-[#28282A] text-white'
                      : 'border-[#262628] bg-[#1C1C1E] text-[#999] hover:border-[#38383A]'
                  }`}
                >
                  <span className="w-3.5 h-3.5 rounded-full border border-black/20" style={{ background: p.color }} />
                  {p.name}
                </button>
              ))}
            </div>
          </div>

          {/* 外框比例 */}
          <div>
            <label className="text-xs text-[#777] uppercase tracking-wider block mb-2">画框构图 (Frame Ratio)</label>
            <div className="grid grid-cols-4 gap-1.5">
              {ASPECT_PRESETS.map((a) => (
                <button
                  key={a.label}
                  onClick={() => setConfig({ ...config, targetAspect: a.value })}
                  className={`py-1.5 text-xs rounded border text-center transition-all ${
                    config.targetAspect === a.value
                      ? 'border-white bg-[#28282A] text-white'
                      : 'border-[#262628] bg-[#1C1C1E] text-[#888] hover:border-[#38383A]'
                  }`}
                >
                  {a.label}
                </button>
              ))}
            </div>
          </div>

          {/* 边距与底边加权 */}
          <div className="space-y-4 pt-2 border-t border-[#222]">
            <div>
              <div className="flex justify-between text-xs text-[#888] mb-1">
                <span>卡纸边距 (Base Margin)</span>
                <span>{Math.round((config.marginRatio || 0.15) * 100)}%</span>
              </div>
              <input
                type="range"
                min="0.08"
                max="0.30"
                step="0.01"
                value={config.marginRatio}
                onChange={(e) => setConfig({ ...config, marginRatio: parseFloat(e.target.value) })}
                className="w-full accent-white bg-[#222] h-1 rounded"
              />
            </div>

            <div>
              <div className="flex justify-between text-xs text-[#888] mb-1">
                <span>底边视觉加权 (Bottom Weighted)</span>
                <span>+{Math.round(((config.bottomWeight || 1.25) - 1) * 100)}%</span>
              </div>
              <input
                type="range"
                min="1.0"
                max="1.7"
                step="0.05"
                value={config.bottomWeight}
                onChange={(e) => setConfig({ ...config, bottomWeight: parseFloat(e.target.value) })}
                className="w-full accent-white bg-[#222] h-1 rounded"
              />
            </div>
          </div>

          {/* 无墨纸张钢印 */}
          <div className="pt-2 border-t border-[#222]">
            <div className="flex items-center justify-between mb-3">
              <label className="text-xs text-[#777] uppercase tracking-wider">纸张钢印 (Blind Deboss)</label>
              <input
                type="checkbox"
                checked={config.enableStamp}
                onChange={(e) => setConfig({ ...config, enableStamp: e.target.checked })}
                className="accent-white cursor-pointer"
              />
            </div>

            {config.enableStamp && (
              <div className="space-y-2">
                <input
                  type="text"
                  placeholder="相机机型 (如 LEICA M6)"
                  value={config.cameraModel || ''}
                  onChange={(e) => setConfig({ ...config, cameraModel: e.target.value })}
                  className="w-full bg-[#1C1C1E] border border-[#28282A] text-xs px-3 py-2 rounded text-white focus:outline-none focus:border-[#555]"
                />
                <input
                  type="text"
                  placeholder="胶卷型号 (如 KODAK PORTRA 400)"
                  value={config.filmBrand || ''}
                  onChange={(e) => setConfig({ ...config, filmBrand: e.target.value })}
                  className="w-full bg-[#1C1C1E] border border-[#28282A] text-xs px-3 py-2 rounded text-white focus:outline-none focus:border-[#555]"
                />
              </div>
            )}
          </div>
        </div>

        {/* 导出按钮 */}
        <div className="pt-4 border-t border-[#222]">
          <button
            onClick={handleExport}
            disabled={!sourceImage || isExporting}
            className={`w-full py-3 text-xs tracking-widest uppercase font-medium rounded transition-all ${
              sourceImage && !isExporting
                ? 'bg-white text-black hover:bg-[#EAEAEA] active:scale-[0.99]'
                : 'bg-[#222] text-[#555] cursor-not-allowed'
            }`}
          >
            {isExporting ? '正在渲染超清印刷大图...' : '无损导出画廊装裱作品 (JPEG)'}
          </button>
        </div>
      </aside>
    </div>
  );
};
```
